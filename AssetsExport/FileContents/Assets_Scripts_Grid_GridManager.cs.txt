using System.Collections.Generic;
using UnityEngine;

public class GridManager : MonoBehaviour
{
    [SerializeField] private float cellSize = 1f;
    [SerializeField] private GameObject cellPrefab;

    private GridCell[,] grid;
    private readonly List<GameObject> spawnedCells = new List<GameObject>();

    public Vector2Int StartCell { get; private set; }
    public Vector2Int GoalCell { get; private set; }

    public void BuildGrid(LevelData levelData)
    {
        ClearGrid();

        StartCell = levelData.startCell;
        GoalCell = levelData.goalCell;

        grid = new GridCell[levelData.width, levelData.height];

        for (int y = 0; y < levelData.height; y++)
        {
            for (int x = 0; x < levelData.width; x++)
            {
                Vector2Int cellPosition = new Vector2Int(x, y);
                Vector3 worldPosition = new Vector3(x * cellSize, y * cellSize, 0f);

                bool isBlocked = levelData.blockedCells.Contains(cellPosition);
                grid[x, y] = new GridCell(x, y, isBlocked);

                GameObject cellObject = CreateCell(cellPosition, worldPosition);
                spawnedCells.Add(cellObject);
            }
        }
    }

    public GridCell GetCell(int x, int y)
    {
        if (grid == null)
            return null;

        if (x < 0 || y < 0 || x >= grid.GetLength(0) || y >= grid.GetLength(1))
            return null;

        return grid[x, y];
    }

    public GridCell GetCell(Vector2Int position)
    {
        return GetCell(position.x, position.y);
    }

    public List<GridCell> GetNeighbors(GridCell cell)
    {
        List<GridCell> neighbors = new List<GridCell>(4);

        TryAddNeighbor(neighbors, cell.x + 1, cell.y);
        TryAddNeighbor(neighbors, cell.x - 1, cell.y);
        TryAddNeighbor(neighbors, cell.x, cell.y + 1);
        TryAddNeighbor(neighbors, cell.x, cell.y - 1);

        return neighbors;
    }

    private void TryAddNeighbor(List<GridCell> neighbors, int x, int y)
    {
        GridCell neighbor = GetCell(x, y);
        if (neighbor != null)
        {
            neighbors.Add(neighbor);
        }
    }

    private GameObject CreateCell(Vector2Int cellPosition, Vector3 worldPosition)
    {
        GameObject cellObject;

        if (cellPrefab != null)
        {
            cellObject = Instantiate(cellPrefab, worldPosition, Quaternion.identity, transform);
        }
        else
        {
            cellObject = new GameObject($"Cell_{cellPosition.x}_{cellPosition.y}");
            cellObject.transform.SetParent(transform);
            cellObject.transform.position = worldPosition;

            SpriteRenderer sr = cellObject.AddComponent<SpriteRenderer>();
            sr.sprite = CreateRuntimeSprite();
        }

        cellObject.name = $"Cell_{cellPosition.x}_{cellPosition.y}";

        SpriteRenderer renderer = cellObject.GetComponent<SpriteRenderer>();
        if (renderer != null)
        {
            renderer.color = GetCellColor(cellPosition);
        }

        return cellObject;
    }

    private Color GetCellColor(Vector2Int cellPosition)
    {
        if (cellPosition == StartCell)
            return Color.green;

        if (cellPosition == GoalCell)
            return Color.red;

        GridCell cell = GetCell(cellPosition);
        if (cell != null && cell.isBlocked)
            return Color.gray;

        return Color.white;
    }

    private void ClearGrid()
    {
        for (int i = 0; i < spawnedCells.Count; i++)
        {
            if (spawnedCells[i] != null)
            {
                Destroy(spawnedCells[i]);
            }
        }

        spawnedCells.Clear();
        grid = null;
    }

    private Sprite CreateRuntimeSprite()
    {
        Texture2D texture = new Texture2D(1, 1);
        texture.SetPixel(0, 0, Color.white);
        texture.Apply();

        return Sprite.Create(
            texture,
            new Rect(0, 0, 1, 1),
            new Vector2(0.5f, 0.5f),
            1f
        );
    }

    public bool WouldBlockPath(Vector2Int cellPosition)
    {
        GridCell cell = GetCell(cellPosition);

        if (cell == null)
            return true;

        bool originalState = cell.isBlocked;

        // Temporär blockieren
        cell.isBlocked = true;

        Pathfinder pathfinder = new Pathfinder(this);
        bool hasPath = pathfinder.HasPath(StartCell, GoalCell);

        // Zustand zurücksetzen
        cell.isBlocked = originalState;

        return !hasPath;
    }

    public bool CanBuildAt(Vector2Int cellPosition)
    {
        GridCell cell = GetCell(cellPosition);

        if (cell == null)
            return false;

        if (cellPosition == StartCell || cellPosition == GoalCell)
            return false;

        if (cell.isBlocked)
            return false;

        bool originalState = cell.isBlocked;
        cell.isBlocked = true;

        Pathfinder pathfinder = new Pathfinder(this);
        bool hasPath = pathfinder.HasPath(StartCell, GoalCell);

        cell.isBlocked = originalState;

        return hasPath;
    }

    public Vector2Int WorldToCell(Vector3 worldPosition)
    {
        int x = Mathf.FloorToInt(worldPosition.x / cellSize);
        int y = Mathf.FloorToInt(worldPosition.y / cellSize);

        return new Vector2Int(x, y);
    }

    public bool TryBlockCell(Vector2Int cellPosition)
    {
        if (!CanBuildAt(cellPosition))
            return false;

        GridCell cell = GetCell(cellPosition);
        if (cell == null)
            return false;

        cell.isBlocked = true;
        return true;
    }

    public void RefreshGridVisuals()
    {
        if (grid == null)
            return;

        for (int i = 0; i < spawnedCells.Count; i++)
        {
            if (spawnedCells[i] == null)
                continue;

            string[] parts = spawnedCells[i].name.Replace("Cell_", "").Split('_');
            if (parts.Length != 2)
                continue;

            int x = int.Parse(parts[0]);
            int y = int.Parse(parts[1]);

            SpriteRenderer renderer = spawnedCells[i].GetComponent<SpriteRenderer>();
            if (renderer != null)
            {
                renderer.color = GetCellColor(new Vector2Int(x, y));
            }
        }
    }
}
